      document.addEventListener('DOMContentLoaded', function () {
        var card = document.getElementById('tcg-status-card');
        var text = document.getElementById('tcg-status-text');
        if (!card) return;
        fetch('/api/status', { headers: { accept: 'application/json' } })
          .then(function (r) { if (!r.ok) throw 0; return r.json(); })
          .then(function (d) {
            var st = (d.overall && d.overall.state) || 'unknown';
            var cls = st === 'ok' ? 'is-ok' : st === 'degraded' ? 'is-degraded' : st === 'down' ? 'is-down' : st === 'maintenance' ? 'is-maint' : '';
            if (cls) card.classList.add(cls);
            var svc = d.services || [];
            var up = svc.filter(function (s) { return s.state === 'ok'; }).length;
            var label = (d.overall && d.overall.label) || 'Status';
            if (svc.length) label += ' · ' + up + '/' + svc.length + ' services operational';
            if (text) text.textContent = label;
          })
          .catch(function () { /* leave the static fallback text as-is */ });
      });
