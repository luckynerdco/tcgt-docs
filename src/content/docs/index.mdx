      document.addEventListener('DOMContentLoaded', function () {
        var card = document.getElementById('tcg-status-card');
        var text = document.getElementById('tcg-status-text');
        var list = document.getElementById('tcg-status-list');
        if (!card) return;
        var KNOWN = { ok: 1, degraded: 1, down: 1, maintenance: 1 };
        fetch('/api/status', { headers: { accept: 'application/json' } })
          .then(function (r) { if (!r.ok) throw 0; return r.json(); })
          .then(function (d) {
            var st = (d.overall && d.overall.state) || 'unknown';
            if (KNOWN[st]) card.classList.add('is-' + st);
            if (text && d.overall && d.overall.label) text.textContent = d.overall.label;
            var svc = d.services || [];
            if (list && svc.length) {
              list.innerHTML = '';
              svc.forEach(function (s) {
                var li = document.createElement('li');
                var dot = document.createElement('span');
                dot.className = 'tcg-status-dot ' + (KNOWN[s.state] ? s.state : 'unknown');
                if (s.label) dot.title = s.label;
                var name = document.createElement('span');
                name.textContent = s.name;
                li.appendChild(dot);
                li.appendChild(name);
                list.appendChild(li);
              });
            }
          })
          .catch(function () { /* leave the static fallback as-is */ });
      });
